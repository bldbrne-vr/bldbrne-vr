cmake_minimum_required(VERSION 3.20)
project(vr_fsr31_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(FFX_VK_PORTABLE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(FFX_VK_PORTABLE_BUILD_PROBE OFF CACHE BOOL "" FORCE)
set(FFX_VK_PORTABLE_BUILD_FSR3_HOST OFF CACHE BOOL "" FORCE)
set(FFX_VK_PORTABLE_BUILD_FSR3_VK_BACKEND OFF CACHE BOOL "" FORCE)
set(FFX_VK_PORTABLE_BUILD_FSR4_V07_VULKAN OFF CACHE BOOL "" FORCE)
set(FFX_VK_PORTABLE_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${FSR31_SOURCE}" provider)
add_library(vr_fsr31_probe SHARED bridge.cpp)
option(FSR31_PRODUCTION "Name the bridge for normal runtime selection" OFF)
if(FSR31_PRODUCTION)
 set_target_properties(vr_fsr31_probe PROPERTIES OUTPUT_NAME vr_fsr31_stereo)
endif()
target_include_directories(vr_fsr31_probe PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../../../src")
target_link_libraries(vr_fsr31_probe PRIVATE ffx-vulkan::fsr3-vk-backend-3.1.5)
target_sources(vr_fsr31_probe PRIVATE catalog.cpp)
target_include_directories(vr_fsr31_probe PRIVATE
 "${FSR31_SOURCE}/upstream/ffx-2.3.0/Kits/FidelityFX/api/internal"
 "${FSR31_SOURCE}/upstream/ffx-2.3.0/Kits/FidelityFX/upscalers/fsr3/include")
# Keep the pinned dependency unchanged. Its layout cache omitted the dependency
# between consecutive clears already in TRANSFER_DST_OPTIMAL (validation WAW).
set(bridge_source "${FSR31_SOURCE}/src/ffx_vk_fsr3_3_1_5_bridge.cpp")
file(READ "${bridge_source}" bridge_text)
set(old_check "    if (resource->imageLayout == desiredLayout)\n        return true;")
set(new_check "    if (resource->imageLayout == desiredLayout) {\n        if (desiredLayout == VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL)\n            transition_image(commandBuffer, resource, desiredLayout, desiredLayout,\n                VK_PIPELINE_STAGE_TRANSFER_BIT, VK_PIPELINE_STAGE_TRANSFER_BIT,\n                VK_ACCESS_TRANSFER_WRITE_BIT, VK_ACCESS_TRANSFER_WRITE_BIT);\n        return true;\n    }")
string(FIND "${bridge_text}" "${old_check}" found)
if(found EQUAL -1)
 message(FATAL_ERROR "Pinned FSR3 bridge layout check changed; review the clear dependency patch")
endif()
string(REPLACE "${old_check}" "${new_check}" bridge_text "${bridge_text}")
include("${CMAKE_CURRENT_SOURCE_DIR}/constant_pool.cmake")
set(patched "${CMAKE_CURRENT_BINARY_DIR}/fsr31_bridge_clear_fixed.cpp")
file(WRITE "${patched}" "${bridge_text}")
get_target_property(provider_sources ffx_vulkan_fsr3_vk_backend_3_1_5 SOURCES)
list(REMOVE_ITEM provider_sources src/ffx_vk_fsr3_3_1_5_bridge.cpp)
set_property(TARGET ffx_vulkan_fsr3_vk_backend_3_1_5 PROPERTY SOURCES "${provider_sources};${patched}")
# LDR permutation generated with build_ldr.py; upstream defaults are HDR-only.
get_filename_component(probe_build_root "${FSR31_SOURCE}" DIRECTORY)
set(ldr_modules "${probe_build_root}/fsr31-ldr")
if(NOT EXISTS "${ldr_modules}/ffx_vk_fsr3_3_1_5_embedded_spirv.h")
 message(FATAL_ERROR "Run build_ldr.py before configuring this LDR-only probe")
endif()
target_include_directories(ffx_vulkan_portable BEFORE PRIVATE "${ldr_modules}")

# CPU-only adapter regression tests use fake SDK/Vulkan calls; no GPU or game required.
option(FSR31_ADAPTER_TESTS "Build the FSR 3.1.5 adapter contract tests" ON)
if(FSR31_ADAPTER_TESTS AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/fsr31_bridge_test.cpp")
 enable_testing()
 add_executable(fsr31_bridge_test ../../tests/fsr31_bridge_test.cpp)
 target_compile_definitions(fsr31_bridge_test PRIVATE NOMINMAX)
 target_include_directories(fsr31_bridge_test PRIVATE
  "${FSR31_SOURCE}/include"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../../../src"
  "${CMAKE_CURRENT_SOURCE_DIR}/../../../../externals/vulkan-headers/include")
 add_test(NAME fsr31_bridge_contract COMMAND fsr31_bridge_test)
endif()
