cmake_minimum_required(VERSION 3.24)
project(BenchmarkXR LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ZLIB_COMPAT ON CACHE BOOL "" FORCE)
set(ZLIB_ENABLE_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_GTEST OFF CACHE BOOL "" FORCE)
get_filename_component(REPO "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
find_package(Vulkan REQUIRED)
add_subdirectory("${REPO}/externals/zlib-ng" zlib EXCLUDE_FROM_ALL)
add_library(benchmark_xr SHARED mockxr.cpp)
target_include_directories(benchmark_xr PRIVATE "${REPO}/src" "${REPO}/externals/openxr/include")
target_link_libraries(benchmark_xr PRIVATE Vulkan::Vulkan zlibstatic)
if(MSVC)
 target_compile_options(benchmark_xr PRIVATE /utf-8 /EHsc)
endif()
